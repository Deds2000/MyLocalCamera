   ```ts

   import { twMerge } from "tailwind-merge";

   export function cn(...inputs: ClassValue[]): string {
     return twMerge(clsx(inputs));
   }
   ```
5. `tailwind-preset.ts` — exporte um objeto `Config` do Tailwind (`import type { Config } from "tailwindcss"`) com `theme.extend`:
   - `colors`: `bg: { DEFAULT, raised }`, `glass: { DEFAULT: "rgb(255 255 255 / 0.05)", strong: "rgb(255 255 255 / 0.09)", border: "rgb(255 255 255 / 0.10)" }`, `accent: { DEFAULT, strong, blue }`, `ok`, `warn`, `danger`, `text: { DEFAULT, muted }`, **mais** as cores semânticas do shadcn mapeadas para as variáveis (`border: "hsl(var(--border))"`, `input`, `ring`, `background`, `foreground`, `primary: { DEFAULT: "hsl(var(--primary))", foreground: "hsl(var(--primary-foreground))" }`, `secondary`, `destructive`, `muted`, `card`, `popover` — mesmo padrão; **atenção:** `accent` do shadcn usa `--accent`, então nomeie a cor de destaque do app como `brand` **além** de `accent` se houver conflito, e registre a decisão em comentário);
   - `borderRadius`: `sm: "8px"`, `md: "12px"`, `lg: "16px"`, `xl: "24px"`;
   - `boxShadow`: `glass`, `overlay`, `"glow-accent"`;
   - `fontFamily`: `sans: ['"Inter Variable"', "Inter", "system-ui", "sans-serif"]`, `mono: ['"JetBrains Mono Variable"', "ui-monospace", "monospace"]`;
   - `keyframes`/`animation`: `pulse-ring` (scale 1→1.8 e opacity .6→0, 1.6s infinite) para o ponto "AO VIVO".
   Exporte `export default preset;` e também `export const tokens = { ... }` (objeto simples com os hex) para teste.
6. `src/styles/tokens.css`: bloco `:root { ...variáveis do shadcn... color-scheme: dark; }` e, depois, `html { background: #070B14; }`.
7. `src/i18n/pt-BR.json` e `en.json` com a mesma estrutura mínima: `{ "common": { "appName": "LocalCamera", "actions": { "cancel": "Cancelar" } } }` (en: `"Cancel"`).
8. `src/i18n/index.ts`:
   ```ts
   import i18n from "i18next";
   import { initReactI18next } from "react-i18next";
   import ptBR from "./pt-BR.json";
   import en from "./en.json";

   export const resources = { "pt-BR": { translation: ptBR }, en: { translation: en } } as const;
   export type AppLanguage = "pt-BR" | "en";

   export function detectLanguage(systemLang: string | undefined): AppLanguage {
     return systemLang?.toLowerCase().startsWith("pt") ? "pt-BR" : "en";
   }

   export async function createI18n(lng: AppLanguage = "pt-BR") {
     const instance = i18n.createInstance();
     await instance.use(initReactI18next).init({
       resources,
       lng,
       fallbackLng: "en",
       interpolation: { escapeValue: false },
     });
     return instance;
   }
   ```
9. `src/index.ts`: `export { cn } from "./lib/cn"; export * from "./i18n";`.
10. `.eslintrc` **não** é necessário (a config plana da raiz cobre). Garanta que `pnpm --filter @localcamera/shared lint` roda (adicione `eslint.config.js` local só se o ESLint reclamar: `import root from "../../eslint.config.js"; export default root;`).
