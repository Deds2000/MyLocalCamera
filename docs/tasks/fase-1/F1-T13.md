## Passos
1. `scripts/soak.ps1` (esqueleto):
   ```powershell
   param([Parameter(Mandatory)][string]$ProcessName, [int]$Minutes = 30, [string]$Out = "soak.csv", [int]$WarmupMinutes = 1)
   $ErrorActionPreference = 'Stop'
   $end = (Get-Date).AddMinutes($Minutes)
   "t,cpu_pct,ws_mb,handles,threads" | Set-Content $Out
   $prev = $null
   while ((Get-Date) -lt $end) {
     $p = Get-Process -Name $ProcessName -ErrorAction SilentlyContinue | Select-Object -First 1
     if (-not $p) { Write-Warning "processo $ProcessName não encontrado"; Start-Sleep 10; continue }
     $now = Get-Date
     if ($prev) { $cpu = (($p.TotalProcessorTime - $prev.cpu).TotalSeconds / ($now - $prev.t).TotalSeconds) / [Environment]::ProcessorCount * 100 } else { $cpu = 0 }
     $prev = @{ cpu = $p.TotalProcessorTime; t = $now }
     "{0:o},{1:N1},{2:N1},{3},{4}" -f $now, $cpu, ($p.WorkingSet64/1MB), $p.HandleCount, $p.Threads.Count | Add-Content $Out
     Start-Sleep 10
   }
   ```
